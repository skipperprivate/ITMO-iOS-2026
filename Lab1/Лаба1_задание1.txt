import Foundation

struct Order {
let id: Int
let customer: String
let amount: Double // сумма заказа
let isPaid: Bool
let promoCode: String? // может отсутствовать
}

let orders: [Order] = [
Order(id: 1, customer: "Анна", amount: 1200, isPaid: true, promoCode: "SALE10"),
Order(id: 2, customer: "Иван", amount: 800, isPaid: false, promoCode: nil),
Order(id: 3, customer: "Анна", amount: 5400, isPaid: true, promoCode: nil),
Order(id: 4, customer: "Олег", amount: 300, isPaid: true, promoCode: "NEW"),
Order(id: 5, customer: "Иван", amount: 2100, isPaid: false, promoCode: "SALE10")
]

//Получить массив сумм всех оплаченных заказов.

let arrPaid = orders.filter { $0.isPaid }.map { $0.amount }
print(arrPaid)

//Найти общую сумму всех неоплаченных заказов (reduce).

let unpaidTotalSum = orders.filter { !$0.isPaid }.reduce(0) { $0 + $1.amount }
print(unpaidTotalSum)

//Получить список уникальных имён клиентов, отсортированный по алфавиту.

let arrUniqueCustomers = Array(Set(orders.map { $0.customer })).sorted()
print(arrUniqueCustomers)

//Получить массив промокодов, исключив nil (compactMap).

let arrPromoCodes = orders.compactMap { $0.promoCode }
print(arrPromoCodes)

//Сгруппировать заказы по имени клиента в словарь [String: [Order]] (через Dictionary(grouping:by:)).

let groupedOrders: [String: [Order]] = Dictionary(grouping: orders, by: { $0.customer })
print(groupedOrders)

//Найти клиента с максимальной суммарной оплаченной суммой. (Обозначили, что id заказа)

let topCustomer = Set(orders.map { $0.customer }).map { customer in (customer, orders.filter { $0.customer == customer && $0.isPaid }.reduce(0) { $0 + $1.amount })}.max { $0.1 < $1.1 }
print(topCustomer!)
